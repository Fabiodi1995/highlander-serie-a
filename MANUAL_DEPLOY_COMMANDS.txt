COMANDI MANUALI DEPLOY E VERIFICA PRODUZIONE
===========================================

ESEGUI SUL SERVER PRODUZIONE:

```bash
# 1. VERIFICA STATO DATABASE
sudo -u postgres psql highlander_prod << 'EOF'
SELECT COUNT(*) as teams FROM teams;
SELECT COUNT(*) as matches FROM matches;
SELECT COUNT(*) as games FROM games;
SELECT COUNT(*) as users FROM users WHERE is_admin = true;
\q
EOF

# 2. POPOLA DATI SERIE A SE MANCANTI
cd /home/highlander/app
node load_all_matches.js

# 3. VERIFICA DATI POPOLATI
sudo -u postgres psql highlander_prod << 'EOF'
SELECT COUNT(*) as teams_after FROM teams;
SELECT COUNT(*) as matches_after FROM matches;
SELECT name FROM teams ORDER BY name LIMIT 10;
\q
EOF

# 4. CONFIGURA NGINX PER WEBSOCKET
sudo tee /etc/nginx/sites-available/highlander_websocket << 'EOF'
location /ws/ {
    proxy_pass http://127.0.0.1:5000;
    proxy_http_version 1.1;
    proxy_set_header Upgrade $http_upgrade;
    proxy_set_header Connection "upgrade";
    proxy_set_header Host $host;
    proxy_set_header X-Real-IP $remote_addr;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    proxy_set_header X-Forwarded-Proto $scheme;
    proxy_read_timeout 86400;
}
EOF

# Aggiorna configurazione nginx principale
sudo sed -i '/location \/ {/i\    include /etc/nginx/sites-available/highlander_websocket;' /etc/nginx/sites-enabled/default
sudo nginx -t
sudo systemctl reload nginx

# 5. COPIA ICONE PWA SE MANCANTI
mkdir -p /home/highlander/app/dist/icons
cp /home/highlander/app/client/public/icons/* /home/highlander/app/dist/icons/ 2>/dev/null || echo "Icone da copiare manualmente"

# 6. AGGIORNA MANIFEST PWA
cp /home/highlander/app/client/public/manifest.json /home/highlander/app/dist/ 2>/dev/null || echo "Manifest da copiare manualmente"

# 7. RIAVVIA SERVIZI
npm run build
pm2 restart highlander
sudo systemctl reload nginx

# 8. VERIFICA FINALE
sleep 10
pm2 status
pm2 logs highlander --lines 5
```
